Who can open and edit the tracker create table if not exists public.editors ( email text primary key ); insert into public.editors (email) values ('vinz@neovara.io'), ('greg@neovara.io'), ('adithya@neovara.io') on conflict do nothing; -- 2. Tracker data: one row per to-do, query, decision, visibility check, target and activity entry create table if not exists public.tracker_items ( kind text not null, id text not null, data jsonb not null, sort integer not null default 0, updated_at timestamptz not null default now(), updated_by text, primary key (kind, id) ); -- 3. Lock both tables down alter table public.editors enable row level security; alter table public.tracker_items enable row level security; -- 4. Helper: is the signed-in user on the editors list? create or replace function public.is_editor() returns boolean language sql stable security definer set search_path = public as $$ select exists ( select 1 from public.editors where lower(email) = lower(coalesce(auth.jwt() ->> 'email', '')) ); $$; grant execute on function public.is_editor() to anon, authenticated; -- 5. Only signed-in people on the editors list can read or change anything drop policy if exists "editors read" on public.tracker_items; drop policy if exists "editors insert" on public.tracker_items; drop policy if exists "editors update" on public.tracker_items; drop policy if exists "editors delete" on public.tracker_items; create policy "editors read" on public.tracker_items for select to authenticated using (public.is_editor()); create policy "editors insert" on public.tracker_items for insert to authenticated with check (public.is_editor()); create policy "editors update" on public.tracker_items for update to authenticated using (public.is_editor()) with check (public.is_editor()); create policy "editors delete" on public.tracker_items for delete to authenticated using (public.is_editor()); -- (No policies on public.editors: nobody can read or change the list from the browser. -- Add or remove people here in the SQL editor or the Table Editor.) -- 6.
